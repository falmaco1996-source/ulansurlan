-- Extensions
create extension if not exists pgcrypto;

-- 1) Admin whitelist table
create table if not exists public.admin_users (
  user_id uuid primary key references auth.users(id) on delete cascade,
  created_at timestamptz not null default now()
);

-- Helper: check admin
create or replace function public.is_admin()
returns boolean
language sql
stable
as $$
  select exists (
    select 1 from public.admin_users au where au.user_id = auth.uid()
  );
$$;

-- 2) Enum status aspirasi
do $$
begin
  if not exists (select 1 from pg_type where typname = 'aspirasi_status') then
    create type public.aspirasi_status as enum
      ('Masuk','Diverifikasi','Diteruskan','Diproses','Selesai','Ditolak');
  end if;
end$$;

-- 3) Ticket sequence + generator
create sequence if not exists public.aspirasi_ticket_seq start 1;

create or replace function public.generate_ticket_code()
returns text
language sql
as $$
  select 'BDG-' || lpad(nextval('public.aspirasi_ticket_seq')::text, 6, '0');
$$;

-- 4) Table aspirasi (PRIVATE base table)
create table if not exists public.aspirasi (
  id uuid primary key default gen_random_uuid(),
  ticket_code text unique not null default public.generate_ticket_code(),

  kategori text not null,
  lokasi text not null,
  isi text not null,

  -- data pelapor (SENSITIF: jangan ditampilkan publik)
  nama text,
  kontak text,

  status public.aspirasi_status not null default 'Masuk',
  catatan_admin text,

  is_public boolean not null default true,

  created_at timestamptz not null default now(),
  updated_at timestamptz not null default now()
);

-- updated_at trigger
create or replace function public.touch_updated_at()
returns trigger
language plpgsql
as $$
begin
  new.updated_at = now();
  return new;
end;
$$;

drop trigger if exists trg_aspirasi_updated_at on public.aspirasi;
create trigger trg_aspirasi_updated_at
before update on public.aspirasi
for each row execute function public.touch_updated_at();

-- 5) Public view (AMAN untuk publik)
create or replace view public.aspirasi_public as
select
  ticket_code,
  kategori,
  lokasi,
  status,
  updated_at,
  created_at
from public.aspirasi
where is_public = true;

do $$
declare
  idx_exists boolean;
  idx_is_unique boolean;
begin
  -- cek apakah index bernama reses_unique_year_masa sudah ada
  select exists (
    select 1
    from pg_class c
    join pg_namespace n on n.oid = c.relnamespace
    where n.nspname = 'public'
      and c.relname = 'reses_unique_year_masa'
      and c.relkind = 'i'
  ) into idx_exists;

  -- kalau index ada, cek dia unique atau bukan
  if idx_exists then
    select i.indisunique
    into idx_is_unique
    from pg_index i
    join pg_class c on c.oid = i.indexrelid
    join pg_namespace n on n.oid = c.relnamespace
    where n.nspname = 'public'
      and c.relname = 'reses_unique_year_masa';
  end if;

  -- kalau index belum ada -> buat unique index
  if not idx_exists then
    execute 'create unique index reses_unique_year_masa on public.reses (tahun, masa)';
  elsif idx_exists and not coalesce(idx_is_unique, false) then
    raise exception 'Index reses_unique_year_masa sudah ada tapi BUKAN unique. Hapus/rename index itu dulu atau ganti nama index baru.';
  end if;

  -- kalau constraint belum ada -> bikin constraint pakai index yang sudah ada (tidak bikin relation/index baru)
  if not exists (select 1 from pg_constraint where conname = 'reses_unique_year_masa') then
    execute 'alter table public.reses add constraint reses_unique_year_masa unique using index reses_unique_year_masa';
  end if;

end $$;

-- 1) Tambah kolom relasi ke reses (kalau belum ada)
alter table public.aspirasi
add column if not exists reses_id uuid references public.reses(id) on delete set null;

create index if not exists aspirasi_reses_id_idx on public.aspirasi(reses_id);

-- 2) Helper: hitung ulang counter aduan di tabel reses
create or replace function public.recalc_reses_aduan_counts(_reses_id uuid)
returns void
language plpgsql
as $$
begin
  update public.reses r
  set
    aduan_masuk = (select count(*) from public.aspirasi a where a.reses_id = _reses_id),
    aduan_selesai = (select count(*) from public.aspirasi a where a.reses_id = _reses_id and a.status = 'Selesai'),
    updated_at = now()
  where r.id = _reses_id;
end;
$$;

-- 3) Trigger: setiap insert/update/delete aspirasi, counter reses ikut update
create or replace function public.trg_aspirasi_reses_counts()
returns trigger
language plpgsql
as $$
begin
  if (tg_op = 'INSERT') then
    if new.reses_id is not null then
      perform public.recalc_reses_aduan_counts(new.reses_id);
    end if;
    return new;

  elsif (tg_op = 'UPDATE') then
    if old.reses_id is distinct from new.reses_id then
      if old.reses_id is not null then
        perform public.recalc_reses_aduan_counts(old.reses_id);
      end if;
      if new.reses_id is not null then
        perform public.recalc_reses_aduan_counts(new.reses_id);
      end if;
    else
      -- reses_id sama, tapi status berubah -> hitung ulang
      if new.reses_id is not null and (old.status is distinct from new.status) then
        perform public.recalc_reses_aduan_counts(new.reses_id);
      end if;
    end if;
    return new;

  elsif (tg_op = 'DELETE') then
    if old.reses_id is not null then
      perform public.recalc_reses_aduan_counts(old.reses_id);
    end if;
    return old;
  end if;

  return null;
end;
$$;

drop trigger if exists aspirasi_reses_counts on public.aspirasi;

create trigger aspirasi_reses_counts
after insert or update or delete on public.aspirasi
for each row execute function public.trg_aspirasi_reses_counts();
